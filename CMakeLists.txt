option(DOOMS_SANITIZE "Build dooms targets with AddressSanitizer and UBSan" OFF)
option(DOOMS_BUILD_TESTS "Build tests and benchmarks" ON)
function(dooms_target tgt)
  target_compile_options(${tgt} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(DOOMS_SANITIZE)
    target_compile_options(${tgt} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
    target_link_options(${tgt} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
